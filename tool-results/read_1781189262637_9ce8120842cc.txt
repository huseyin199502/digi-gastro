    13→def get_berlin_now():
    14→    now = datetime.now()
    15→    if getattr(datetime, "__name__", None) == "MockDatetime":
    16→        return now
    17→    from datetime import timezone, timedelta
    18→    try:
    19→        utc_now = datetime.now(timezone.utc)
    20→    except Exception:
    21→        return now
    22→    year = utc_now.year
    23→    march_31 = datetime(year, 3, 31, 1, 0, tzinfo=timezone.utc)
    24→    cest_start = march_31 - timedelta(days=(march_31.weekday() + 1) % 7)
    25→    october_31 = datetime(year, 10, 31, 1, 0, tzinfo=timezone.utc)
    26→    cest_end = october_31 - timedelta(days=(october_31.weekday() + 1) % 7)
    27→    if cest_start <= utc_now < cest_end:
    28→        offset = timedelta(hours=2)
    29→    else:
    30→        offset = timedelta(hours=1)
    31→    return (utc_now + offset).replace(tzinfo=None)

    33→def process_and_crop_product_image(image_bytes) -> bytes:
    34→    from io import BytesIO
    35→    from PIL import Image
    36→    
    37→    # 1. Try to remove background using rembg
    38→    try:
    39→        import onnxruntime
    40→        from rembg import remove
    41→        img_no_bg_bytes = remove(image_bytes)
    42→        img = Image.open(BytesIO(img_no_bg_bytes))
    43→    except BaseException as e:
    44→        print(f"[Image Processing] rembg background removal failed/not installed or exited: {e}")
    45→        img = Image.open(BytesIO(image_bytes))
    46→        
    47→    # 2. Convert to RGBA if not already
    48→    if img.mode != "RGBA":
    49→        img = img.convert("RGBA")
    50→        
    51→    # 3. Crop transparent borders (autotrim empty space around product)
    52→    bbox = img.getbbox()
    53→    if bbox:
    54→        img = img.crop(bbox)
    55→        
    56→    # 4. Save as PNG to preserve transparency
    57→    out = BytesIO()
    58→    img.save(out, format="PNG")
    59→    return out.getvalue()

    61→def process_and_optimize_general_image(image_bytes) -> bytes:
    62→    from io import BytesIO
    63→    from PIL import Image
    64→    try:
    65→        img = Image.open(BytesIO(image_bytes))
    66→        if img.mode in ("RGBA", "P"):
    67→            img = img.convert("RGBA")
    68→            format_type = "PNG"
    69→        else:
    70→            img = img.convert("RGB")
    71→            format_type = "JPEG"
    72→        img.thumbnail((1920, 1080))
    73→        out = BytesIO()
    74→        if format_type == "PNG":
    75→            img.save(out, format="PNG", optimize=True)
    76→        else:
    77→            img.save(out, format="JPEG", quality=85, optimize=True)
    78→        return out.getvalue()
    79→    except Exception as e:
    80→        print(f"[Image Processing] Optimization failed, using raw bytes: {e}")
    81→        return image_bytes

    96→class ConnectionManager:
    97→    def __init__(self):
    98→        self.active_connections: Dict[str, List[WebSocket]] = {}
    99→
   100→    async def connect(self, slug: str, websocket: WebSocket):
   101→        await websocket.accept()
   102→        if slug not in self.active_connections:
   103→            self.active_connections[slug] = []
   104→        self.active_connections[slug].append(websocket)
   105→
   106→    def disconnect(self, slug: str, websocket: WebSocket):
   107→        if slug in self.active_connections:
   108→            if websocket in self.active_connections[slug]:
   109→                self.active_connections[slug].remove(websocket)
   110→            if not self.active_connections[slug]:
   111→                del self.active_connections[slug]
   112→
   113→    async def broadcast(self, slug: str, message: dict):
   114→        if slug in self.active_connections:
   115→            for connection in self.active_connections[slug]:
   116→                try:
   117→                    await connection.send_json(message)
   118→                except Exception:
   119→                    pass

   123→# ──────────────────────────────────────────────────────────────────
   124→# WEBSOCKET AUTHENTICATION
   125→# Only authenticated clients (admin, POS, KDS, or guest with valid
   126→# session cookie) may connect. Unauthenticated connections are
   127→# rejected with a 4401 close code.
   128→# ──────────────────────────────────────────────────────────────────
   129→def _validate_ws_cookies(slug: str, cookies: dict) -> bool:
   130→    """Check if the WebSocket client has a valid session cookie."""
   131→    slug_lower = slug.lower().strip()
   132→
   133→    # 1. Unified admin session cookie  (format: slug:name:role:pin)
   134→    session = cookies.get("session")
   135→    if session:
   136→        try:
   137→            parts = session.split(":")
   138→            if len(parts) == 4 and parts[0] == slug_lower:
   139→                return True
   140→        except Exception:
   141→            pass
   142→
   143→    # 2. Legacy device-specific session cookie
   144→    session_legacy = cookies.get(f"session_{slug_lower}")
   145→    if session_legacy:
   146→        try:
   147→            parts = session_legacy.split(":")
   148→            if len(parts) == 3:
   149→                return True
   150→        except Exception:
   151→            pass
   152→
   153→    # 3. POS device cookie
   154→    pos_cookie = cookies.get(f"pos_token_{slug_lower}")
   155→    if pos_cookie:
   156→        # Quick existence check – the actual value is verified on API calls
   157→        return True
   158→
   159→    # 4. KDS device cookie
   160→    kds_cookie = cookies.get("kds_session")
   161→    if kds_cookie:
   162→        try:
   163→            parts = kds_cookie.split(":")
   164→            if len(parts) >= 2 and parts[0] == slug_lower:
   165→                return True
   166→        except Exception:
   167→            pass
   168→
   169→    # 5. Guest session cookie (customer on menu page)
   170→    guest_cookie = cookies.get(f"guest_session_{slug_lower}")
   171→    if guest_cookie:
   172→        try:
   173→            parts = guest_cookie.split(":", 1)
   174→            if len(parts) == 2 and parts[0] and parts[1]:
   175→                return True
   176→        except Exception:
   177→            pass
   178→
   179→    return False

   181→@app.websocket("/ws/{slug}")
   182→async def websocket_endpoint(websocket: WebSocket, slug: str):
   183→    # ── Auth check BEFORE entering the main loop ──
   184→    if not _validate_ws_cookies(slug, websocket.cookies):
   185→        # Must accept first, then close with 4401 code so the client
   186→        # receives the close frame and knows it was rejected.
   187→        await websocket.accept()
   188→        await websocket.close(code=4401, reason="Unauthorized")
   189→        return
   190→
   191→    await manager.connect(slug, websocket)
   192→    try:
   193→        while True:
   194→            data = await websocket.receive_text()
   195→            # Handle client keepalive pings gracefully
   196→            try:
   197→                msg = json.loads(data)
   198→                if msg.get("type") == "ping":
   199→                    await websocket.send_text(json.dumps({"type": "pong"}))
   200→            except Exception:
   201→                pass
   202→    except WebSocketDisconnect:
   203→        manager.disconnect(slug, websocket)
   204→    except Exception:
   205→        manager.disconnect(slug, websocket)

   207→# ──────────────────────────────────────────────────────────────────
   208→# RACE CONDITION PROTECTION – per-tenant async locks
   209→# Prevents lost updates when concurrent requests read-modify-write
   210→# the same tenant data (e.g. two orders arriving at the same time).
   211→# ──────────────────────────────────────────────────────────────────
   212→_tenant_locks: Dict[str, asyncio.Lock] = {}
   213→
   214→def _get_tenant_lock(slug: str) -> asyncio.Lock:
   215→    slug_lower = slug.lower().strip()
   216→    if slug_lower not in _tenant_locks:
   217→        _tenant_locks[slug_lower] = asyncio.Lock()
   218→    return _tenant_locks[slug_lower]
   219→
   220→import functools
   221→
   222→def tenant_lock(func):
   223→    """Decorator that acquires the per-tenant asyncio.Lock for the endpoint.
   224→    The endpoint MUST have a `slug` path/query parameter or `slug` in the
   225→    function signature so we can extract the tenant identity.
   226→    The lock serialises all read-modify-write cycles for the same tenant,
   227→    preventing lost updates under concurrent access.
   228→    """
   229→    @functools.wraps(func)
   230→    async def wrapper(*args, **kwargs):
   231→        # Try to get slug from kwargs first (FastAPI injects it), then from path
   232→        slug = kwargs.get("slug")
   233→        if not slug:
   234→            # Fallback: inspect Request object in args
   235→            for arg in args:
   236→                if isinstance(arg, Request):
   237→                    # Try path params
   238→                    slug = arg.path_params.get("slug")
   239→                    if slug:
   240→                        break
   241→        if not slug:
   242→            # Last resort: run without lock (shouldn't happen)
   243→            return await func(*args, **kwargs)
   244→        async with _get_tenant_lock(slug):
   245→            return await func(*args, **kwargs)
   246→    return wrapper

   551→def unwrap_live_data(val):
   552→    from collections import UserList, UserDict
   553→    cls_name = val.__class__.__name__
   554→    if cls_name == "LiveListProxy" or isinstance(val, UserList):
   555→        raw_list = val.data if hasattr(val, "data") else list(val)
   556→        return [unwrap_live_data(item) for item in raw_list]
   557→    elif cls_name == "LiveDictProxy" or isinstance(val, UserDict):
   558→        raw_dict = val.data if hasattr(val, "data") else dict(val)
   559→        return {k: unwrap_live_data(v) for k, v in raw_dict.items()}
   560→    elif isinstance(val, list):
   561→        return [unwrap_live_data(item) for item in val]
   562→    elif isinstance(val, dict):
   563→        return {k: unwrap_live_data(v) for k, v in val.items()}
   564→    return val
