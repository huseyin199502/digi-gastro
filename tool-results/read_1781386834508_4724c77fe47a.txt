   100→   100→    async def connect(self, slug: str, websocket: WebSocket):
   101→   101→        await websocket.accept()
   102→   102→        if slug not in self.active_connections:
   103→   103→            self.active_connections[slug] = []
   104→   104→        self.active_connections[slug].append(websocket)
   105→   105→
   106→   106→    def disconnect(self, slug: str, websocket: WebSocket):
   107→   107→        if slug in self.active_connections:
   108→   108→            if websocket in self.active_connections[slug]:
   109→   109→                self.active_connections[slug].remove(websocket)
   110→   110→            if not self.active_connections[slug]:
   111→   111→                del self.active_connections[slug]

   124→   124→# WEBSOCKET AUTHENTICATION
   125→   125→# Only authenticated clients (admin, POS, KDS, or guest with valid

   129→   129→def _validate_ws_cookies(slug: str, cookies: dict) -> bool:
   130→   130→    """Check if the WebSocket client has a valid session cookie."""
   131→   131→    slug_lower = slug.lower().strip()
   132→   132→
   133→   133→    # 1. Unified admin session cookie  (format: slug:name:role:pin)
   134→   134→    session = cookies.get("session")
   135→   135→    if session:
   136→   136→        try:
   137→   137→            parts = session.split(":")
   138→   138→            if len(parts) == 4 and parts[0] == slug_lower:
   139→   139→                return True
   140→   140→        except Exception:
   141→   141→            pass
   142→   142→
   143→   143→    # 2. Legacy device-specific session cookie
   144→   144→    session_legacy = cookies.get(f"session_{slug_lower}")
   145→   145→    if session_legacy:
   146→   146→        try:
   147→   147→            parts = session_legacy.split(":")
   148→   148→            if len(parts) == 3:
   149→   149→                return True
   150→   150→        except Exception:
   151→   151→            pass
   152→   152→
   153→   153→    # 3. POS device cookie
   154→   154→    pos_cookie = cookies.get(f"pos_token_{slug_lower}")
   155→   155→    if pos_cookie:
   156→   156→        # Quick existence check – the actual value is verified on API calls
   157→   157→        return True
   158→   158→
   159→   159→    # 4. KDS device cookie
   160→   160→    kds_cookie = cookies.get("kds_session")
   161→   161→    if kds_cookie:
   162→   162→        try:
   163→   163→            parts = kds_cookie.split(":")
   164→   164→            if len(parts) >= 2 and parts[0] == slug_lower:
   165→   165→                return True
   166→   166→        except Exception:
   167→   167→            pass
   168→   168→
   169→   169→    # 5. Guest session cookie (customer on menu page) - validate slug exists in DB
   170→   170→    guest_cookie = cookies.get(f"guest_session_{slug_lower}")
   171→   171→    if guest_cookie:
   172→   172→        try:
   173→   173→            parts = guest_cookie.split(":", 1)
   174→   174→            if len(parts) == 2 and parts[0] and parts[1]:
   175→   175→                # Verify the tenant slug actually exists to prevent random connections
   176→   176→                try:
   177→   177→                    db_check = SessionLocal()
   178→   178→                    tenant_exists = db_check.query(Tenant).filter(Tenant.slug == slug_lower).first() is not None
   179→   179→                    db_check.close()
   180→   180→                    if tenant_exists:
   181→   181→                        return True
   182→   182→                except Exception:
   183→   183→                    pass
   184→   184→        except Exception:
   185→   185→            pass
   186→   186→
   187→   187→    return False

   189→   189→@app.websocket("/ws/{slug}")
   190→   190→async def websocket_endpoint(websocket: WebSocket, slug: str):
   191→   191→    # ── Auth check BEFORE entering the main loop ──
   192→   192→    if not _validate_ws_cookies(slug, websocket.cookies):
   193→   193→        # Must accept first, then close with 4401 code so the client
   194→   194→        # receives the close frame and knows it was rejected.
   195→   195→        await websocket.accept()
   196→   196→        await websocket.close(code=4401, reason="Unauthorized")
   197→   197→        return
   198→   198→
   199→   199→    await manager.connect(slug, websocket)
   200→   200→    try:
   201→   201→        while True:
   202→   202→            data = await websocket.receive_text()
   203→   203→            # Handle client keepalive pings gracefully
   204→   204→            try:
   205→   205→                msg = json.loads(data)
   206→   206→                if msg.get("type") == "ping":
   207→   207→                    await websocket.send_text(json.dumps({"type": "pong"}))
   208→   208→            except Exception:
   209→   209→                pass
   210→   210→    except WebSocketDisconnect:
   211→   211→        manager.disconnect(slug, websocket)
   212→   212→    except Exception:
   213→   213→        manager.disconnect(slug, websocket)

   215→   215→# ──────────────────────────────────────────────────────────────────
   216→   216→# RACE CONDITION PROTECTION – per-tenant async locks
   217→   217→# Prevents lost updates when concurrent requests read-modify-write
   218→   218→# the same tenant data (e.g. two orders arriving at the same time).
   219→   219→# ──────────────────────────────────────────────────────────────────
   220→   220→_tenant_locks: Dict[str, asyncio.Lock] = {}
   221→   221→
   222→   222→def _get_tenant_lock(slug: str) -> asyncio.Lock:
   223→   223→    slug_lower = slug.lower().strip()
   224→   224→    if slug_lower not in _tenant_locks:
   225→   225→        _tenant_locks[slug_lower] = asyncio.Lock()
   226→   226→    return _tenant_locks[slug_lower]
   227→   227→
   228→   228→import functools
   229→   229→
   230→   230→def tenant_lock(func):
   231→   231→    """Decorator that acquires the per-tenant asyncio.Lock for the endpoint.
   232→   232→    The endpoint MUST have a `slug` path/query parameter or `slug` in the
   233→   233→    function signature so we can extract the tenant identity.
   234→   234→    The lock serialises all read-modify-write cycles for the same tenant,
   235→   235→    preventing lost updates under concurrent access.
   236→   236→    """
   237→   237→    @functools.wraps(func)
   238→   238→    async def wrapper(*args, **kwargs):
   239→   239→        # Try to get slug from kwargs first (FastAPI injects it), then from path
   240→   240→        slug = kwargs.get("slug")
   241→   241→        if not slug:
   242→   242→            # Fallback: inspect Request object in args
   243→   243→            for arg in args:
   244→   244→                if isinstance(arg, Request):
   245→   245→                    # Try path params
   246→   246→                    slug = arg.path_params.get("slug")
   247→   247→                    if slug:
   248→   248→                        break
   249→   249→        if not slug:
   250→   250→            # Last resort: run without lock (shouldn't happen)
   251→   251→            return await func(*args, **kwargs)
   252→   252→        async with _get_tenant_lock(slug):
   253→   253→            return await func(*args, **kwargs)
   254→   254→    return wrapper

   605→   605→def unwrap_live_data(val):
   606→   606→    from collections import UserList, UserDict
   607→   607→    cls_name = val.__class__.__name__
   608→   608→    if cls_name == "LiveListProxy" or isinstance(val, UserList):
   609→   609→        raw_list = val.data if hasattr(val, "data") else list(val)
   610→   610→        return [unwrap_live_data(item) for item in raw_list]
   611→   611→    elif cls_name == "LiveDictProxy" or isinstance(val, UserDict):
   612→   612→        raw_dict = val.data if hasattr(val, "data") else dict(val)
   613→   613→        return {k: unwrap_live_data(v) for k, v in raw_dict.items()}
   614→   614→    elif isinstance(val, list):
   615→   615→        return [unwrap_live_data(item) for item in val]
   616→   616→    elif isinstance(val, dict):
   617→   617→        return {k: unwrap_live_data(v) for k, v in val.items()}
   618→   618→    return val
