  7632→            
  7633→            // Requires a minimum horizontal swipe length of 80px, and horizontal shift must be greater than vertical
  7634→            if (Math.abs(diffX) > 80 && Math.abs(diffX) > Math.abs(diffY)) {
  7635→                const currentTab = localStorage.getItem('dl_admin_active_tab') || 'live';
  7636→                const currentIndex = tabsSequence.indexOf(currentTab);
  7637→                
  7638→                if (diffX < 0) {
  7639→                    // Swiped left -> Switch to next tab

  7641→                        switchAdminTab(tabsSequence[currentIndex + 1]);
